@import view.{Order, Product, Routes, Delivery, Products} @( ordersData: Vector[Order] = Vector(), productsData: Vector[Product] = Vector(), product: Option[Product], deliveriesData: Vector[Delivery] = Vector(), delivery: Option[Delivery], )(implicit request: Request[AnyContent])
@{ if(request.path == Routes.OrdersPath) { orders.content(data = ordersData) } else if(request.path == Routes.ProductsPath) { products.content(data = productsData) } else if(request.path.contains("product/edit")) { products.edit(data = productsData, product.get) } else if(request.path.contains("product/create")) { products.create(data = productsData, product.get) } else if(request.path.contains("warehouse/create")) { deliveries.create(deliveriesData = deliveriesData, delivery = delivery.get, productsData = productsData) } else if(request.path == Routes.DeliveriesPath) { deliveries.content(data = deliveriesData) } else { dashboard.content() } }
